file(GLOB_RECURSE VECTORIZATION_LIST ${CMAKE_CURRENT_LIST_DIR}/*.cpp)
set(VECTORIZATION_TARGET vectorization)
add_library(${VECTORIZATION_TARGET} ${VECTORIZATION_LIST})

option(OMNI_ENABLE_ISAL_CRYPTO_MD5 "Auto-detect ISA-L Crypto for vectorized MD5" ON)
if(OMNI_ENABLE_ISAL_CRYPTO_MD5)
    set(OMNI_ISAL_CRYPTO_MD5_FOUND FALSE)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(ISAL_CRYPTO QUIET IMPORTED_TARGET libisal_crypto)
    endif()

    if(TARGET PkgConfig::ISAL_CRYPTO)
        target_link_libraries(${VECTORIZATION_TARGET} PRIVATE PkgConfig::ISAL_CRYPTO)
        set(OMNI_ISAL_CRYPTO_MD5_FOUND TRUE)
    else()
        find_path(ISAL_CRYPTO_MD5_INCLUDE_DIR NAMES isa-l_crypto/md5_mb.h)
        find_library(ISAL_CRYPTO_MD5_LIBRARY NAMES isal_crypto libisal_crypto)
        if(ISAL_CRYPTO_MD5_INCLUDE_DIR AND ISAL_CRYPTO_MD5_LIBRARY)
            target_include_directories(${VECTORIZATION_TARGET} PRIVATE ${ISAL_CRYPTO_MD5_INCLUDE_DIR})
            target_link_libraries(${VECTORIZATION_TARGET} PRIVATE ${ISAL_CRYPTO_MD5_LIBRARY})
            set(OMNI_ISAL_CRYPTO_MD5_FOUND TRUE)
        endif()
    endif()

    if(OMNI_ISAL_CRYPTO_MD5_FOUND)
        target_compile_definitions(${VECTORIZATION_TARGET} PUBLIC OMNI_HAVE_ISAL_CRYPTO_MD5=1)
        message(STATUS "ISA-L Crypto detected: vectorized MD5 acceleration enabled")
    else()
        message(STATUS "ISA-L Crypto not detected: keeping the scalar MD5 implementation")
    endif()
else()
    message(STATUS "ISA-L Crypto MD5 detection disabled: keeping the scalar MD5 implementation")
endif()

find_package(fmt REQUIRED)
find_package(double-conversion REQUIRED)
find_package(folly REQUIRED)
find_package(gflags REQUIRED)
if(NOT TARGET gflags::gflags)
    # This is a bit convoluted, but we want to be able to use gflags::gflags as a
    # target even when omni is built as a subproject which uses
    # `find_package(gflags)` which does not create a globally imported target that
    # we can ALIAS.
    add_library(gflags_gflags INTERFACE)
    target_link_libraries(gflags_gflags INTERFACE gflags)
    add_library(gflags::gflags ALIAS gflags_gflags)
endif()

target_link_libraries(${VECTORIZATION_TARGET} PRIVATE util expression udf re2::re2 fmt::fmt double-conversion::double-conversion Folly::folly gflags::gflags)
file(GLOB EXPRESSION_HEAD_FILES ${SOURCE_ROOT}/src/vectorization/*.h)
install(FILES ${EXPRESSION_HEAD_FILES} DESTINATION ${CMAKE_INSTALL_PREFIX}/include/vectorization)

file(GLOB REGISTRATION_HEAD_FILES ${SOURCE_ROOT}/src/vectorization/registration/*.h)
